from contextlib import ExitStack, suppress
from types import TracebackType
from typing import reveal_type


class SwallowZero:
    def __enter__(self) -> int:
        return 7

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc: BaseException | None,
        tb: TracebackType | None,
    ) -> bool:
        return exc_type is ZeroDivisionError


def compute(divisor: int) -> int:
    with SwallowZero() as v:
        result = v // divisor
    return result


def safe_div(a: int, b: int) -> int | None:
    out: int | None = None
    with suppress(ZeroDivisionError):
        out = a // b
    reveal_type(out)
    return out


def stacked() -> list[str]:
    log: list[str] = []
    with ExitStack() as stack:
        stack.callback(log.append, "closed")
        log.append("body")
    reveal_type(log)
    return log


print(compute(2))
print(safe_div(9, 0), safe_div(9, 3))
print(stacked())
try:
    print(compute(0))
except UnboundLocalError:
    print("UnboundLocalError: swallowed exception left result unset")
